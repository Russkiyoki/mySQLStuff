-- ### JOINS AND WHAT NOT ### --
Example tables:
Students:
ID   Name
1    Alex
2    Sam
3    Jordan 

Enrollments
StudentId    Class
1            Math
1            Science
2            History

-- inner join
SELECT Students.Name, Enrollments.Class 
FROM Students
INNER JOIN Enrollments ON Students.ID = Enrollments.StudentID;

Result
Name    Class
Alex    Math
Alex    Science
Sam     History


-- Left Join example shows everything shows every name

SELECT Students.Name, Enrollments.Class
FROM Students
LEFT JOIN Enrollments ON Students.ID = Enrollments.StudentId;

Result 

Name    class
Alex    Science
Sam     History
Jordan  Null


-- right Join example
-- shows everything from enrollments.class's
SELECT Students.Name, Enrollments.Class
FROM Students
RIGHT JOIN Enrollments ON Students.ID Enrollments.StudentID;

Result
Name   class
Alex   Math
Alex   Science
Sam    History
NULL   geography
