-- List all products showing product name, categoryID, price, and quantity
-- on hand. 
SELECT ProductName, CategoryID, UnitPrice, Qoh FROM Products;

-- Retrieve all products that are out of stock ( quantity = 0)
SELECT * FROM Products WHERE Qoh = 0;

-- Display products priced between $100 and $500
-- Revision 1: changed "UnitPrice > 100 AND UnitPrice < 500" to 
-- "BETWEEN 100 AND 500". This changed was recommended by the Grader
-- Using BETWEEN instead of the mathematical signs improves readability
-- and also effect because theres a lower chance of me missing numbers. 
SELECT * FROM Products WHERE UnitPrice BETWEEN 100 AND 500;

-- Show the total number of products group by category
-- I am selecting the categoryName and putting a count on each record
-- and when it get printed out it will show me category and then
-- totalProducts, and it will print out totals per categoryName. 
-- Revision 1: changed from CategoryName to CategoryID, since Products
-- does not have CategoryName :/
SELECT CategoryID, COUNT(*) AS totalProducts FROM Products GROUP BY CategoryID; 

-- Print average price of products per category using GROUP BY
-- AVG() returns the average of all entries for the column provided
-- Then we group by the category name 
SELECT CategoryID, AVG(UnitPrice) AS avgPrice FROM Products GROUP BY CategoryID;

-- Join Products and Suppliers to list product Name, supplier name,
-- and contact details.
SELECT Products.ProductName, Suppliers.SupplierName, Suppliers.Phone, Suppliers.Email
FROM Products 
INNER JOIN Suppliers ON Suppliers.SupplierID = Products.SupplierID;

-- join products categories and Orders to display productName, categoryName,
-- quantitySold, and date of sale
-- In my original project I had four another table of OrderDetails to adhere to 3nf

SELECT Products.ProductName, Categories.CategoryName, OrderDetails.Quantity, Orders.OrderDate 
FROM Orders
INNER JOIN OrderDetails ON Orders.OrderID = OrderDetails.OrderID
INNER JOIN Products ON OrderDetails.ProductID = Products.ProductID
INNER JOIN Categories ON Products.CategoryID = Categories.CategoryID;

-- Use a left join to list all suppliers and any products they supply
-- This will also print out suppliers who dont supply anything
SELECT Suppliers.SupplierName, Products.ProductName
FROM Suppliers
LEFT JOIN Products ON Suppliers.SupplierID = Products.SupplierID;

-- Full Outer Join
-- Full Outer Join can be implemented by having a Left join unioned
-- with a right join. 
SELECT Products.ProductName, Suppliers.SupplierName
FROM Products
LEFT JOIN Suppliers ON Products.SupplierID = Suppliers.SupplierID
UNION
SELECT Products.ProductName, Suppliers.SupplierName
FROM Products
RIGHT JOIN Suppliers ON Products.SupplierID = Suppliers.SupplierID;

-- use a HAVING clause to find all categories with more than 10 products in stock
-- Originally, I counted all rows and had a counter then group by each category.
-- The grader has instructed to first get rid of my spelling mistakes. That has been
-- fixed. Then to calculate the total sum of Qoh instead of counting. 
-- I had forgotten that Qoh existed.
SELECT Categories.CategoryName, SUM(Products.Qoh) AS TotalStock
FROM Products
INNER JOIN Categories ON Products.CategoryID = Categories.CategoryID
GROUP BY Categories.CategoryName
HAVING SUM(Products.Qoh) > 10;
